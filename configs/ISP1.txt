! ===== ISP1 - deliberately NO security (public addressing, simple static routing) =====
enable
configure terminal
hostname ISP1
interface GigabitEthernet0/0
 description To HO-R1
 ip address 200.1.1.2 255.255.255.252
 no shutdown
interface GigabitEthernet0/1
 description To ISP2
 ip address 200.2.2.1 255.255.255.252
 no shutdown
ip route 200.3.3.0 255.255.255.252 200.2.2.2
end
write memory
