! ===== HO-R1 (Cisco 2911) - Head Office router: router-on-a-stick, DHCP, ACLs, AAA, SSH, IPsec =====
! Packet Tracer: enable the crypto feature first, save, reload:
!   license boot module c2900 technology-package securityk9
!   do write memory  ->  do reload
enable
configure terminal
hostname HO-R1
no ip domain-lookup
ip domain-name walchand.local
enable secret WtcEna2026
service password-encryption
banner motd ^ AUTHORIZED ACCESS ONLY - Walchand Tech Campus HO. Activity is monitored. ^
username admin privilege 15 secret WtcAdm2026
!
! ---- AAA (RADIUS first, local fallback) ----
aaa new-model
radius-server host 192.168.40.10 key Rad1us2026
aaa authentication login default group radius local
aaa authorization exec default group radius local
!
! ---- Interfaces / inter-VLAN routing ----
interface GigabitEthernet0/0
 description Trunk to HO-SW1
 no shutdown
interface GigabitEthernet0/0.10
 description ADMIN
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
interface GigabitEthernet0/0.20
 description STAFF
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
 ip access-group STAFF-IN in
interface GigabitEthernet0/0.30
 description GUEST
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
 ip access-group GUEST-IN in
interface GigabitEthernet0/0.40
 description SERVER
 encapsulation dot1Q 40
 ip address 192.168.40.1 255.255.255.0
interface GigabitEthernet0/0.99
 description MGMT
 encapsulation dot1Q 99
 ip address 192.168.99.1 255.255.255.0
!
interface GigabitEthernet0/1
 description To ISP1 (untrusted)
 ip address 200.1.1.1 255.255.255.252
 no shutdown
ip route 0.0.0.0 0.0.0.0 200.1.1.2
!
! ---- DHCP ----
ip dhcp excluded-address 192.168.10.1 192.168.10.20
ip dhcp excluded-address 192.168.20.1 192.168.20.20
ip dhcp excluded-address 192.168.30.1 192.168.30.20
ip dhcp pool ADMIN
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 192.168.40.10
ip dhcp pool STAFF
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 192.168.40.10
ip dhcp pool GUEST
 network 192.168.30.0 255.255.255.0
 default-router 192.168.30.1
 dns-server 192.168.40.10
!
! ---- ACLs (inter-VLAN + Guest control) ----
ip access-list extended GUEST-IN
 permit udp any eq bootpc any eq bootps
 permit tcp 192.168.30.0 0.0.0.255 host 192.168.40.10 eq www
 permit udp 192.168.30.0 0.0.0.255 host 192.168.40.10 eq domain
 deny ip 192.168.30.0 0.0.0.255 192.168.0.0 0.0.255.255 log
 deny ip 192.168.30.0 0.0.0.255 172.16.0.0 0.0.255.255 log
 permit ip any any
ip access-list extended STAFF-IN
 deny ip 192.168.20.0 0.0.0.255 192.168.10.0 0.0.0.255 log
 deny ip 192.168.20.0 0.0.0.255 192.168.99.0 0.0.0.255 log
 permit ip any any
ip access-list standard MGMT-ONLY
 permit 192.168.99.0 0.0.0.255
 permit 192.168.10.0 0.0.0.255
!
! ---- Secure management: SSHv2 only ----
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3
no ip http server
line con 0
 logging synchronous
 exec-timeout 5 0
line vty 0 4
 transport input ssh
 access-class MGMT-ONLY in
 exec-timeout 5 0
!
! ---- Site-to-site IPsec VPN to BO (200.3.3.1) ----
crypto isakmp policy 10
 encryption aes 256
 hash sha
 authentication pre-share
 group 2
 lifetime 86400
! (comparison run: change to "encryption 3des" or "encryption des", then compare "show crypto isakmp policy")
crypto isakmp key WalchandVPN2026 address 200.3.3.1
crypto ipsec transform-set WTC-TS esp-aes 256 esp-sha-hmac
 mode tunnel
access-list 110 permit ip 192.168.0.0 0.0.255.255 172.16.10.0 0.0.0.255
crypto map WTC-MAP 10 ipsec-isakmp
 set peer 200.3.3.1
 set transform-set WTC-TS
 match address 110
interface GigabitEthernet0/1
 crypto map WTC-MAP
end
write memory
