! ===== HO-SW1 (Cisco 2960) - distribution: trunks, server, APs =====
enable
configure terminal
hostname HO-SW1
no ip domain-lookup
ip domain-name walchand.local
enable secret WtcEna2026
service password-encryption
banner motd ^ AUTHORIZED ACCESS ONLY ^
username admin privilege 15 secret WtcAdm2026
aaa new-model
radius-server host 192.168.40.10 key Rad1us2026
aaa authentication login default group radius local
!
vlan 10
 name ADMIN
vlan 20
 name STAFF
vlan 30
 name GUEST
vlan 40
 name SERVER
vlan 99
 name MGMT
vlan 999
 name BLACKHOLE
!
interface Vlan99
 ip address 192.168.99.2 255.255.255.0
 no shutdown
ip default-gateway 192.168.99.1
!
! Trunks (native VLAN moved off VLAN 1 -> anti VLAN-hopping)
interface GigabitEthernet0/1
 description Trunk to HO-R1
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,40,99
 switchport nonegotiate
 ip dhcp snooping trust
interface GigabitEthernet0/2
 description Trunk to HO-SW2
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,40,99
 switchport nonegotiate
 ip dhcp snooping trust
!
interface FastEthernet0/1
 description AAA/Web/Mail server
 switchport mode access
 switchport access vlan 40
 spanning-tree portfast
 spanning-tree bpduguard enable
interface FastEthernet0/2
 description AP-STAFF (WPA2-Enterprise)
 switchport mode access
 switchport access vlan 20
interface FastEthernet0/3
 description AP-GUEST (WPA2-PSK)
 switchport mode access
 switchport access vlan 30
!
! Unused ports -> shutdown + blackhole VLAN
interface range FastEthernet0/4 - 24
 switchport mode access
 switchport access vlan 999
 shutdown
!
ip dhcp snooping
ip dhcp snooping vlan 10,20,30
no ip dhcp snooping information option
!
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
line con 0
 exec-timeout 5 0
line vty 0 15
 transport input ssh
 exec-timeout 5 0
end
write memory
